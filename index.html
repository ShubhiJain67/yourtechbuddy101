// LLD: pick a pricing strategy at runtime
interface PricingStrategy {
    int price(int hours);
}

class HourlyPricing implements PricingStrategy {
    public int price(int h) { return h * 40; }
}

class Main {
    public static void main(String[] a) {
        PricingStrategy p = new HourlyPricing();
        System.out.println("Fee: ₹" + p.price(3));
    }
}
$ java Main
Fee: ₹120
